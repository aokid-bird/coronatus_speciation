# This script intends to plot figures related to PCA and Admixture analyses
# These analyses were conducted with different subsets of samples, specifically, those retaining or
# excluding Kanto wintering samples and those retaining or excluding outgroup samples.
# Figure 1 will present only the representative ones, while the others were saved as supplementary
# figures.
# For the Figure 1, the empty space will be used for geographic map that is manually added to the 
# figure with reorganizing the sublabels.

#===========================#
#### Libraries & Sources ####
#===========================#

library(magrittr)

library(colorblindr)

subdir <- 
  expand.grid(category = c("all", "breed"),
              outgrp   = c("", "_wout")) %>% 
  mutate(subdir = str_c(category, outgrp)) %>% 
  pull(subdir)

resdir_angsd_global.local <- "res/angsd_global_rlrmv_mindepthind2"
figdir <- "figures/revise"
#============================#

#============================#
# prepare a list of vectors
pc <- e <- plot.pc <- log <- df.admix <- plot.deltak <- plot.admix <- vector("list", length(subdir))

levels.pop <- c("Tomakomai", "Kushiro", "Russia", "Aomori", "Kanto (wintering)","Outgroup")
levels.poplabel <- c("Tomakomai", "Kushiro", "Russia", "Aomori", "Kanto (wintering)",
                     expression(italic("Laterallus jamaicensis")), 
                     expression(italic("Atlantisia rogersi")))
levels.reg <- c("tomakomai_b", "kushiro_b", 
                "russia_b", "aomori_b", "kanto_w",
                "Laterallus jamaicensis", "Atlantisia rogersi")

condition <- function(x)case_when(
  x == "all" ~ "Both breeding and wintering",
  x == "breed" ~ "Only breeding",
  x == "all_wout" ~ "Breeding and wintering with outgroups",
  x == "breed_wout" ~ "Breeding with outgroups"
)

for(i in 1:length(subdir)){
  ##### 1-1) variable preps #####
  # for this loop set
  tmp.subdir <- subdir[i]
  
  # directories
  resdir_pca <- str_interp("${resdir_angsd_global.local}/${tmp.subdir}/pca")
  pcaout <- str_interp("${resdir_pca}/pcangsd")
  logout <- str_interp("${resdir_pca}/pcangsd.oe")
  name.bamlist <- str_interp("bamlist_${tmp.subdir}")
  resdir_admix <- str_interp("${resdir_angsd_global.local}/${tmp.subdir}/admixture")
  
  # base df for population info
  # CAUTION: minind differs between different bamlist sets
  base.df <- 
    read_delim(str_interp("${resdir_angsd_global.local}/${name.bamlist}"), delim = " ", col_names = FALSE) %>% 
    set_colnames("bamlist.fullpath") %>% 
    mutate(sample = str_extract(bamlist.fullpath, "(?<=/)[^/]+(?=\\.bam)")) %>% 
    left_join(., data.df, by = "sample") %>% 
    left_join(., ref.df, by = c("sample" = "sraid")) %>% 
    mutate(group_plot = case_when(
      is.na(group_reg) ~ species.y,
      TRUE ~ as.character(group_reg)
    )) %>% 
    mutate(group_plot = factor(group_plot, levels = levels.reg))
  #### 1-2) PCA data load ####
  pc[[i]] <- 
    read.table(str_interp("${pcaout}.cov")) %>% 
    as.matrix %>% 
    eigen %>% 
    .$vectors %>% 
    as_tibble %>% 
    set_colnames(str_c("pc",1:ncol(.)))
  # eigen values to percentage explanation
  e[[i]] <- 
    read.table(str_interp("${resdir_pca}/pcangsd.tree.cov")) %>% 
    as.matrix %>% 
    eigen %>% 
    .$values %>% 
    tibble(pc = str_c("pc",1:length(.)), values = .) %>% 
    mutate(pct = round(values/sum(values)*100, digits = 2))
  
  # color palette for pca
  base.colpal <- c(as.character(color("bright")(6))[c(2,4,1,3,5)], "grey60", "grey80")
  sub.colpal <- base.colpal[levels(base.df$group_plot) %in% unique(base.df$group_plot)]
  
  #### 1-3) PCA plot ####
  plot.pc[[i]] <- 
    pcaplot(xaxis = "pc1", yaxis = "pc2", 
            .group = base.df$group_plot, .pc = pc[[i]], .e = e[[i]],
            .colpal = sub.colpal, .labels = levels.poplabel)
  
  #### 1-4) Admixture data load ####
  # log files
  log[[i]] <- 
    list.files(resdir_admix, pattern = ".log", full.names = TRUE) %>% 
    lapply(readLines) %>% 
    sapply(function(x){x[which(str_sub(x, 1, 1) == 'b')]}) %>% 
    as.data.frame() %>% 
    rename(like = ".") %>% 
    mutate(K = list.files(resdir_admix, pattern = ".log")) %>% 
    mutate(like = str_extract(like, "(?<=like=)[0-9.-]+") %>% as.numeric) %>%
    mutate(K = str_extract(K, "(?<=K)[:digit:]")) %>% 
    mutate(runnum =  list.files(resdir_admix, pattern = ".log") %>% str_remove(".log"))
  
  cond <- condition(tmp.subdir)
  
  # plot deltaK changes
  plot.deltak[[i]] <- 
    log[[i]] %>% 
    group_by(K) %>% 
    summarise(deltaK = mean(abs(like))/sd(abs(like))) %>%
    dplyr::slice(-1) %>% 
    ggplot(data = ., aes(x = as.numeric(K), y = deltaK))+
    geom_line() + 
    xlab("K") +
    ylab("delta K") +
    ggtitle(str_interp("${cond}")) +
    theme_bw()
  
  # select the best run for each K to plot
  like <- log[[i]] %>% group_by(K) %>% filter(like == max(like)) %>% slice(1)
  
  for(K in 2:5){
    tmp.admix <- 
      read.table(str_c(resdir_admix, "/", like$runnum[K], ".qopt")) %>% 
      bind_cols(base.df, .) %>% 
      arrange(group_plot) %>% 
      select(sample, group_plot, contains("V", ignore.case = FALSE))
    
    # reorder genetic groups based on population
    order <-
      tmp.admix %>% 
      group_by(group_plot) %>% 
      select(starts_with("V")) %>% 
      summarize_all(mean) %>% 
      ungroup %>% 
      # identify the V cluster scoring the max proportion values for each population
      gather(clust, cnt, starts_with("V")) %>% 
      group_by(group_plot) %>% 
      slice(which.max(cnt)) %>% 
      # arrange the df according to the group order to order the V clusters
      mutate(reordercol = factor(group_plot, levels = c("tomakomai_b", "Laterallus jamaicensis", "Atlantisia rogersi", "russia_b", "kushiro_b", "aomori_b"))) %>% 
      arrange(reordercol) %>% 
      select(!reordercol) %>% 
      pull(clust) %>% 
      unique
    
    order <- c(order, str_c("V", 1:K)[!(str_c("V", 1:K) %in% order)])
    
    tmp.admix %<>% 
      select(all_of(c("sample", "group_plot", order))) %>% 
      set_colnames(c("sample", "group_plot", str_c("V", 1:length(order)))) %>% 
      pivot_longer(cols = starts_with("V"),
                   names_to = "popGroup",
                   names_prefix = "Group.",
                   values_to = "prob",
                   values_drop_na = FALSE) %>% 
      mutate(.K = K)
    
    df.admix[[i]] <- bind_rows(df.admix[[i]], tmp.admix)
  }
  
  ####  1-5) Admixture plot ####
  if(str_detect(tmp.subdir, "wout")){
    admix.colpal <- c(as.character(color("muted")(9)[c(1)]),
                      "#4D4D4D",
                      as.character(color("muted")(9)[c(2,3,4,6,5)]))
  }else{
    admix.colpal <- as.character(color("muted")(9)[c(1,2,3,4,6,5)])
  }
  
  
  plot.admix[[i]] <- 
    df.admix[[i]] %>% 
    mutate(group_plot = str_remove(group_plot, "_.$") %>% str_to_sentence) %>% 
    mutate(group_plot = if_else(group_plot %in% c("Laterallus jamaicensis", "Atlantisia rogersi"), "Outgroup", group_plot)) %>% 
    mutate(Population = if_else(group_plot == "Kanto", "Kanto (wintering)", group_plot) %>% factor(., levels = levels.pop)) %>% 
    mutate(.K = str_c("K = ", .K)) %>% 
    ggplot(., aes(factor(sample), prob, fill = factor(popGroup))) + 
    geom_col(color = "grey", size = 0.1) + 
    labs(x = "Individuals", y = "Ancestry") +
    facet_grid(cols = vars(Population),rows = vars(.K), switch = "x", scales = "free", space = "free") + 
    scale_y_continuous(expand = c(0,0)) +
    scale_x_discrete(expand = expansion(add = 0.6)) +
    scale_fill_manual(values = admix.colpal) +
    theme_minimal() +
    theme(
      panel.spacing.y = unit(0.5, "lines"),
      axis.text.x = element_blank(),
      strip.text.x = element_text(size = 15),
      strip.text.y = element_text(size = 15),
      axis.title.x = element_blank(),
      axis.title.y = element_text(size = 15),
      legend.position = "none"
    )
}

# pcaplot
pcaplot <- function(xaxis, yaxis, .group, .pc, .e, .colpal, .labels, .title = NULL){
  
  # combine into one dataframe
  plot.pc <- 
    .pc %>% 
    dplyr::select(all_of(c(xaxis, yaxis))) %>% 
    mutate(group = .group) %>% 
    set_colnames(c("xaxis","yaxis","group"))
  
  # fix the xaxis range always to have positives as larger absolute values
  xaxis.range <- range(plot.pc$xaxis)
  
  if(abs(xaxis.range[1]) > abs(xaxis.range[2])){
    plot.pc$xaxis <- -plot.pc$xaxis
  }
  
  # fix the yaxis range always to have positives as larger absolute values
  yaxis.range <- range(plot.pc$yaxis)
  
  if(abs(yaxis.range[1]) > abs(yaxis.range[2])){
    plot.pc$yaxis <- -plot.pc$yaxis
  }
  
  # percentage of the total variance
  plot.pct <-
    .e %>% 
    filter(pc %in% c(xaxis, yaxis)) %>% 
    pull(pct)
  
  # plot label
  plot.axis <- str_to_upper(c(xaxis, yaxis))
  
  # plot
  p <-
    ggplot(data = plot.pc, aes(x = xaxis, y = yaxis, color = group))+
      geom_hline(aes(yintercept = 0.00)) + 
      geom_vline(aes(xintercept = 0.00)) + 
      geom_point() + 
      # ggrepel::geom_label_repel(aes(label = sample, color = group_reg), max.overlaps = 15) +
      xlab(str_interp("${plot.axis[1]} (${plot.pct[1]}%)")) + 
      ylab(str_interp("${plot.axis[2]} (${plot.pct[2]}%)")) + 
      stat_ellipse(level=0.5) + 
      scale_color_manual(values = .colpal, labels = .labels) +
      theme_bw() 
  
  if(!is.null(.title)){
    p <- p + ggtitle(.title)
  }
  
  p
}
